#include <stdio.h>

void main(){

const int N=2;

int vetor[N],i=0,soma=0;
float media;

printf("Introduza 10 numeros inteiros\n");

for(i=0;i<N;i++){
    printf("Introduza o numero %d:",i+1);
    scanf("%d", &vetor[i]);
    soma=soma+vetor[i];
}
    media=(float)soma/N;
    printf("\nA media introduzida dos %d introduzidos e de %.2f\n",N+1,media);

for(i=0;i<N;i++){
    if(vetor[i]<media)
        vetor[i]=0;
}

for(i=0;i<N;i++){
    printf("%d ",vetor[i]);
}

}
