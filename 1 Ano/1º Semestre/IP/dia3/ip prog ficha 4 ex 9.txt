#include <stdio.h>

void main(){

int num,max=0,pos=0,total=0;

    printf("Introduza a sua sequencia de numeros em que 0 acaba a sequencia:\n");
    do{
        scanf("%d", &num);

        if (num>0){
            total=total+1;
            if(num>max){
                max=num;
                pos=total;
            }
        }
    }while(num!=0);
    printf("Numero maximo foi %d na posicao %d de um total de %d numeros",max,pos,total);
}
