Select sp.nome "NOME SPONSOR", r.nome "nome circuito", to_char(r.dataHora, "DD-MM-YYYY") "Data Corrida", p.nome "Nome Piloto", st.total "TempoTotal"
from race r
join standing st on st.idRace=r.idRace
join sponsor sp on  sp.idSponsor=r.idSponsor
join pilot p on p.idSponsor=sp.idSponsor
where r.dataHora betweem to_char((sysdate,'YYYY')-10) and to_char(sysdate,'YYYY')
and to_lower(sp.areaDeNegocio) like 'seguro'
order by sp.nome desc, r.dataHora desc;

2.

Select t.nome "NOME DA EQUIPA", sum(st.numPontos) "Num Total de Pontos"
from team t
join pilot p on p.idTeam=t.idTeam
join standing st on st.idPilot=p.idPilot
join race r on r.idRace=st.idRace
where to_lower(t.pais) like 'alemanha'
and where r.dataHora betweem to_char((sysdate,'YYYY')-1) and to_char(sysdate,'YYYY')
group by t.nome
having count(r.idRace) > 4
order by sum(st.numPontos) asc;
